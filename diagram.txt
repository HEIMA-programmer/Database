skinparam linetype ortho

' ==========================================
' 第一层: 产品与供应商
' ==========================================

class "Track" as TRACK {
    +TrackID : int <<PK>>
    +ReleaseID : int <<FK>>
    --
    +TrackNumber : int
    +Title : varchar(255) {NN}
    +Duration : varchar(10)
}

class "ReleaseAlbum" as RELEASE {
    +ReleaseID : int <<PK>>
    --
    +Title : varchar(255) {NN}
    +ArtistName : varchar(255) {NN}
    +LabelName : varchar(255) {NN}
    +ReleaseYear : varchar(4)
    +Genre : varchar(50)
    +Format : varchar(50)
    +Description : text
    +BaseUnitCost : decimal(10,2) {Default 25.00}
}

class "Supplier" as SUPPLIER {
    +SupplierID : int <<PK>>
    --
    +Name : varchar(255) {NN}
    +Email : varchar(255)
}

' ==========================================
' 第二层: 人员与组织
' ==========================================

class "MembershipTier" as TIER {
    +TierID : int <<PK>>
    --
    +TierName : varchar(50)
    +MinPoints : int
    +DiscountRate : decimal(3,2)
}

class "Customer" as CUSTOMER {
    +CustomerID : int <<PK>>
    +TierID : int <<FK>>
    --
    +Name : varchar(255) {NN}
    +Email : varchar(255) {UK}
    +PasswordHash : varchar(255)
    +Birthday : date
    +Points : int {Default 0}
    +CurrentSessionID : varchar(128)
}

class "Shop" as SHOP {
    +ShopID : int <<PK>>
    --
    +Name : varchar(100) {NN}
    +Address : text
    +Type : enum('Retail', 'Warehouse') {NN}
}

class "Employee" as EMPLOYEE {
    +EmployeeID : int <<PK>>
    +ShopID : int <<FK>>
    --
    +Role : enum('Admin', 'Manager', 'Staff') {NN}
    +Name : varchar(100) {NN}
    +Username : varchar(50) {UK}
    +PasswordHash : varchar(255)
    +HireDate : date
    +CurrentSessionID : varchar(128)
}

' ==========================================
' 第三层: 采购进货
' ==========================================

class "SupplierOrder" as SUPPORDER {
    +SupplierOrderID : int <<PK>>
    +SupplierID : int <<FK>>
    +CreatedByEmployeeID : int <<FK>>
    +DestinationShopID : int <<FK>>
    --
    +OrderDate : datetime
    +Status : enum('Pending', 'Received', 'Cancelled')
    +ReceivedDate : datetime
    +TotalCost : decimal(10,2)
}

class "SupplierOrderLine" as SUPPLINE {
    +SupplierOrderID : int <<PK, FK>>
    +ReleaseID : int <<PK, FK>>
    --
    +Quantity : int
    +UnitCost : decimal(10,2)
    +ConditionGrade : enum('New','Mint','NM','VG+','VG')
    +SalePrice : decimal(10,2) {Nullable}
}

class "BuybackOrder" as BUYORDER {
    +BuybackOrderID : int <<PK>>
    +CustomerID : int <<FK, Nullable>>
    +ProcessedByEmployeeID : int <<FK>>
    +ShopID : int <<FK>>
    --
    +BuybackDate : datetime
    +Status : enum('Pending', 'Completed', 'Cancelled')
    +TotalPayment : decimal(10,2)
    +Notes : text
}

class "BuybackOrderLine" as BUYLINE {
    +BuybackOrderID : int <<PK, FK>>
    +ReleaseID : int <<PK, FK>>
    --
    +Quantity : int
    +UnitPrice : decimal(10,2)
    +ConditionGrade : enum('New','Mint','NM','VG+','VG')
}

' ==========================================
' 第四层: 库存管理
' ==========================================

class "StockItem" as STOCKITEM {
    +StockItemID : int <<PK>>
    +ReleaseID : int <<FK>>
    +ShopID : int <<FK>>
    --
    +SourceType : enum('Supplier', 'Buyback')
    +SourceOrderID : int
    +BatchNo : varchar(50) {NN}
    +ConditionGrade : enum('New','Mint','NM','VG+','VG')
    +Status : enum('Available', 'Sold', 'Reserved', 'InTransit')
    +UnitPrice : decimal(10,2)
    +AcquiredDate : datetime
    +DateSold : datetime
}

class "InventoryTransfer" as TRANSFER {
    +TransferID : int <<PK>>
    +StockItemID : int <<FK>>
    +FromShopID : int <<FK>>
    +ToShopID : int <<FK>>
    +AuthorizedByEmployeeID : int <<FK>>
    +ReceivedByEmployeeID : int <<FK, Nullable>>
    --
    +TransferDate : datetime
    +Status : enum('Pending', 'InTransit', 'Completed', 'Cancelled')
    +ReceivedDate : datetime
}

' ==========================================
' 第五层: 销售与申请
' ==========================================

class "CustomerOrder" as CUSTORDER {
    +OrderID : int <<PK>>
    +CustomerID : int <<FK, Nullable>>
    +FulfilledByShopID : int <<FK>>
    +ProcessedByEmployeeID : int <<FK, Nullable>>
    --
    +OrderDate : datetime
    +TotalAmount : decimal(10,2)
    +OrderStatus : enum('Pending', 'Paid', 'Shipped',
    .. 'ReadyForPickup', 'Completed', 'Cancelled')
    +OrderType : enum('InStore', 'Online')
    +FulfillmentType : enum('Shipping', 'Pickup')
    +ShippingAddress : text
    +ShippingCost : decimal(10,2)
}

class "OrderLine" as ORDERLINE {
    +OrderID : int <<PK, FK>>
    +StockItemID : int <<PK, FK>>
    --
    +PriceAtSale : decimal(10,2)
}

class "ManagerRequest" as MGRREQ {
    +RequestID : int <<PK>>
    +RequestedByEmployeeID : int <<FK>>
    +RespondedByEmployeeID : int <<FK, Nullable>>
    +FromShopID : int <<FK>>
    +ToShopID : int <<FK, Nullable>>
    +ReleaseID : int <<FK>>
    --
    +RequestType : enum('PriceAdjustment', 'TransferRequest')
    +ConditionGrade : enum('New','Mint','NM','VG+','VG')
    +Quantity : int
    +CurrentPrice : decimal(10,2)
    +RequestedPrice : decimal(10,2)
    +Reason : text
    +Status : enum('Pending', 'Approved', 'Rejected')
    +AdminResponseNote : text
    +ViewedByRequesterAt : datetime
    +CreatedAt : datetime
    +UpdatedAt : datetime
}

' ==========================================
' 布局控制 (Hidden Layout Hints)
' ==========================================

' 第一层水平排列
TRACK -[hidden]r- RELEASE
RELEASE -[hidden]r- SUPPLIER

' 第一层到第二层
RELEASE -[hidden]d- SHOP
SUPPLIER -[hidden]d- EMPLOYEE

' 第二层水平排列
TIER -[hidden]r- CUSTOMER
CUSTOMER -[hidden]r- SHOP
SHOP -[hidden]r- EMPLOYEE

' 第二层到第三层
CUSTOMER -[hidden]d- BUYORDER
EMPLOYEE -[hidden]d- SUPPORDER

' 第三层水平排列
SUPPORDER -[hidden]r- SUPPLINE
SUPPLINE -[hidden]r- BUYORDER
BUYORDER -[hidden]r- BUYLINE

' 第三层到第四层
SUPPORDER -[hidden]d- STOCKITEM
BUYORDER -[hidden]d- STOCKITEM

' 第四层水平排列
STOCKITEM -[hidden]r- TRANSFER

' 第四层到第五层
STOCKITEM -[hidden]d- ORDERLINE

' 第五层水平排列
CUSTORDER -[hidden]r- ORDERLINE
ORDERLINE -[hidden]r- MGRREQ

' ==========================================
' 关系定义 (Relationships)
' ==========================================

' === 第一层内部 ===
RELEASE "1" -l- "0..*" TRACK : contains >

' === 第二层内部 ===
TIER "1" -r- "0..*" CUSTOMER : classifies >
SHOP "1" -r- "0..*" EMPLOYEE : employs >

' === 第一层 -> 第三层 ===
SUPPLIER "1" -d- "0..*" SUPPORDER : supplies >
RELEASE "1" -d- "0..*" SUPPLINE : ordered in >
RELEASE "1" -d- "0..*" BUYLINE : bought back >

' === 第二层 -> 第三层 ===
EMPLOYEE "1" -d- "0..*" SUPPORDER : creates >
SHOP "1" -d- "0..*" SUPPORDER : receives >
CUSTOMER "1" -d- "0..*" BUYORDER : sells back >
EMPLOYEE "1" -d- "0..*" BUYORDER : processes >
SHOP "1" -d- "0..*" BUYORDER : handles >

' === 第三层内部 ===
SUPPORDER "1" -r- "1..*" SUPPLINE : contains >
BUYORDER "1" -r- "1..*" BUYLINE : contains >

' === 第三层 -> 第四层 ===
SUPPORDER "1" .d. "0..*" STOCKITEM : generates >(Supplier)
BUYORDER "1" .d. "0..*" STOCKITEM : generates >(Buyback)

' === 第一/二层 -> 第四层 ===
RELEASE "1" -d- "0..*" STOCKITEM : instantiates >
SHOP "1" -d- "0..*" STOCKITEM : stores >

' === 第四层内部 ===
STOCKITEM "1" -r- "0..*" TRANSFER : moves >
EMPLOYEE "1" -d- "0..*" TRANSFER : authorizes >

' === 第四层 -> 第五层 ===
STOCKITEM "1" -d- "0..*" ORDERLINE : sold as >

' === 第五层内部 ===
CUSTORDER "1" -r- "1..*" ORDERLINE : contains >

' === 第二层 -> 第五层 ===
CUSTOMER "1" -d- "0..*" CUSTORDER : places >
SHOP "1" -d- "0..*" CUSTORDER : fulfills >
EMPLOYEE "1" -d- "0..*" CUSTORDER : processes >
EMPLOYEE "1" -d- "0..*" MGRREQ : requests >
SHOP "1" -d- "0..*" MGRREQ : from/to >
RELEASE "1" -d- "0..*" MGRREQ : references >

' 图例 (Legend)
' ==========================================
legend right
  |= 符号 |= 含义 |
  | <<PK>> | 主键 |
  | <<FK>> | 外键 |
  | {NN} | NOT NULL |
  | {UK} | UNIQUE |
  | 1 -- 0..* | 一对多 |
  | 1 -- 1..* | 一对多(至少一个) |
  | --- | 实线：直接外键关系 |
  | ... | 虚线：逻辑关系(无直接FK) |
endlegend
